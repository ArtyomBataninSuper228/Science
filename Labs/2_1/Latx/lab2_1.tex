\subsection {Оценка неопределённости измерений}
Неопределённость измерений рассчитывалась согласно классу точности прибора и цене деления шкалы:
\begin{equation}
\begin{aligned}
 \Delta U  = k_{\text{точности}} \cdot U_{\text{Шкалы}} + \frac{\delta U}{2}\\
  \Delta R  = k_{\text{точности}} \cdot R_{\text{Шкалы}} + \frac{\delta R}{2}
 \label{neopr}
 \end{aligned}
\end{equation}

\subsection {Рассчет значений}
\begin{equation}
\begin{aligned}
R_{2+v}  = \frac{1}{\frac{1}{R_2} + \frac{1}{R_v}}\\
 U_{\text{рассч}}  = U_0 \cdot \frac{R_{2+v}}{R_{2+v} + R_1}
 \label{rast}
 \end{aligned}
\end{equation}
